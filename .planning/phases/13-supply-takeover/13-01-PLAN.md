Create `CreationFlowAdapter` class implementing `SupplyTableAdapter` per D-04. Structure:

Constructor signature:
```typescript
constructor(
  onRowsChange: (rows: SupplyRow[]) => void,
  searchFn: (type: SupplyType, query: string) => Promise<SupplySearchResult[]>,
  createFn: (type: SupplyType, data: CreateSupplyData) => Promise<SupplySearchResult>,
)
```

Internal state: `private rows: Map<string, SupplyRow>` (single flat map, keyed by client UUID).

Methods:
- `addThread(threadId, stitchCount, need)`: Check if `threadId` already in rows (duplicate guard). Generate `crypto.randomUUID()` as row id. Create `SupplyRow` with `type: "THREAD"`, `have: 0`, `isNeedOverridden: false`. Store in map. Call `onRowsChange(this.getRows())`. Return `{ success: true, id }`.
- `addBead(beadId, quantity, need)`: Same pattern, `type: "BEAD"`, `stitchCount: 0`.
- `addSpecialty(itemId, need)`: Same pattern, `type: "SPECIALTY"`, `stitchCount: 0`.
- `updateQuantity(type, junctionId, field, value)`: Find row by id in map. Update field. Call onRowsChange. Return success.
- `remove(type, junctionId)`: Delete from map. Call onRowsChange. Return success.
- `searchSupplies(type, query)`: Delegate to `this.searchFn(type, query)`.
- `createSupply(type, data)`: Delegate to `this.createFn(type, data)`.
- `getRows(): SupplyRow[]`: Return `Array.from(this.rows.values())`.
- `loadRows(rows: SupplyRow[]): void`: Clear map, populate from array, call onRowsChange. Used for draft restore.

The `searchFn` and `createFn` are injected so the adapter stays testable without mocking server actions directly. In production, the caller passes `getThreads`/`getBeads`/`getSpecialtyItems` and `createThread`/`createBead`/`createSpecialtyItem` wrappers.

For duplicate detection: before adding, check if any existing row has the same `supplyId` AND same `type`. If found, return `{ success: false, error: "Supply already added" }`.

The adapter stores `code`, `name`, `brandName`, `hexColor` from the search result or create result. The caller (useSupplyTable hook) will call `searchSupplies` to get the SupplySearchResult, then call `addThread`/`addBead`/`addSpecialty`. But the adapter doesn't have access to the search result in the add methods -- it only receives IDs and numbers. So the adapter needs a way to resolve supply metadata. Two options:
- Option A: Store a `supplyMetadata` cache populated during `searchSupplies` calls
- Option B: Accept metadata as additional params

Use Option A matching the LocalStateAdapter pattern: maintain a `private supplyCache: Map<string, SupplySearchResult>` that gets populated on every `searchSupplies` and `createSupply` call. When `addThread` is called with a `threadId`, look up metadata from cache.

If the supply is not in cache (edge case), use fallback values: `code: ""`, `name: "Unknown"`, `brandName: ""`, `hexColor: "#000000"`.

export async function createChartWithSupplies(formData: unknown, supplyPayload: unknown) {
  const user = await requireAuth();

  try {
    const validated = chartFormSchema.parse(formData);
    const supplies = batchSupplySchema.parse(supplyPayload);
    const { chart, project } = validated;

    // Same effectiveStitchCount logic as createChart
    let effectiveStitchCount = chart.stitchCount;
    let effectiveApproximate = chart.stitchCountApproximate;
    if (effectiveStitchCount === 0 && chart.stitchesWide > 0 && chart.stitchesHigh > 0) {
      effectiveStitchCount = chart.stitchesWide * chart.stitchesHigh;
      effectiveApproximate = true;
    }

    const created = await prisma.$transaction(async (tx) => {
      // 1. Create chart + project (same as createChart)
      const result = await tx.chart.create({ data: { ... }, include: { project: true, designer: true, genres: true } });

      // 2. Link fabric (same as createChart)
      if (project.fabricId && result.project) { ... }

      // 3. Batch insert supply junction records (NEW per D-06/D-07)
      const projectId = result.project!.id;

      if (supplies.threads.length > 0) {
        await tx.projectThread.createMany({
          data: supplies.threads.map(t => ({
            projectId,
            threadId: t.supplyId,
            stitchCount: t.stitchCount,
            quantityRequired: t.need,
            quantityAcquired: 0,
            isNeedOverridden: t.isNeedOverridden,
          })),
          skipDuplicates: true,
        });
      }

      if (supplies.beads.length > 0) {
        await tx.projectBead.createMany({
          data: supplies.beads.map(b => ({
            projectId,
            beadId: b.supplyId,
            quantityRequired: b.need,
            quantityAcquired: 0,
          })),
          skipDuplicates: true,
        });
      }

      if (supplies.specialty.length > 0) {
        await tx.projectSpecialty.createMany({
          data: supplies.specialty.map(s => ({
            projectId,
            specialtyItemId: s.supplyId,
            quantityRequired: s.need,
            quantityAcquired: 0,
          })),
          skipDuplicates: true,
        });
      }

      return result;
    });

    // Thumbnail generation (outside transaction, same as createChart)
    let thumbnailWarning: string | undefined;
    if (chart.coverImageUrl) {
      try { await generateThumbnail(created.id, chart.coverImageUrl); }
      catch (err) { console.error(...); thumbnailWarning = "..."; }
    }

    revalidatePath("/charts");
    revalidatePath("/fabric");
    return { success: true as const, chartId: created.id, warning: thumbnailWarning };
  } catch (error) {
    if (error instanceof z.ZodError) {
      return { success: false as const, error: error.errors[0].message };
    }
    console.error("createChartWithSupplies error:", error);
    return { success: false as const, error: "Failed to create chart" };
  }
}

```typescript
export function saveDraftV2(form: ChartFormValues, supplies: SupplyRow[], calcParams: CalcParams): void {
  try {
    const draft: DraftV2 = { version: 2, form, supplies, calcParams };
    localStorage.setItem(DRAFT_KEY, JSON.stringify(draft));
  } catch { /* localStorage full or unavailable */ }
}

export function loadDraftV2(
  defaults: ChartFormValues,
  validDesignerIds: string[],
  validStorageIds: string[],
  validAppIds: string[],
  validFabricIds: string[] = [],
): { form: ChartFormValues; supplies: SupplyRow[]; calcParams: CalcParams } | null {
  try {
    const raw = localStorage.getItem(DRAFT_KEY);
    if (!raw) return null;
    const parsed = JSON.parse(raw);

    // V1 format (no version field) -- wrap in v2 shape
    if (!parsed.version) {
      const form = loadDraft(defaults, validDesignerIds, validStorageIds, validAppIds, validFabricIds);
      if (!form) return null;
      return { form, supplies: [], calcParams: DEFAULT_CALC_PARAMS };
    }

    // V2 format
    const draft = parsed as DraftV2;
    const merged: ChartFormValues = { ...defaults, ...draft.form };

    // Stale ID detection (same pattern as existing loadDraft)
    if (merged.designerId && !validDesignerIds.includes(merged.designerId)) merged.designerId = null;
    if (merged.storageLocationId && !validStorageIds.includes(merged.storageLocationId)) merged.storageLocationId = null;
    if (merged.stitchingAppId && !validAppIds.includes(merged.stitchingAppId)) merged.stitchingAppId = null;
    if (merged.fabricId && !validFabricIds.includes(merged.fabricId)) merged.fabricId = null;

    return {
      form: merged,
      supplies: draft.supplies ?? [],
      calcParams: draft.calcParams ?? DEFAULT_CALC_PARAMS,
    };
  } catch {
    return null;
  }
}
```
