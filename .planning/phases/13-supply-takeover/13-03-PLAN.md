**3. Adapter instantiation (once, via ref):**
```typescript
if (!adapterRef.current) {
  const searchFn = async (type: SupplyType, query: string): Promise<SupplySearchResult[]> => {
    if (type === "THREAD") return getThreads(undefined, undefined, query);
    if (type === "BEAD") return getBeads(query);
    return getSpecialtyItems(query);
  };
  const createFn = async (type: SupplyType, data: CreateSupplyData): Promise<SupplySearchResult> => {
    // Delegate to catalog-only server actions (NOT createAndAdd* which creates junction records)
    if (type === "THREAD") {
      const result = await createThread({ name: data.name, colorCode: data.code, brandId: data.brandId, hexColor: data.hexColor });
      if (!result.success) throw new Error(result.error);
      return { id: result.thread.id, type: "THREAD", code: result.thread.colorCode, name: result.thread.colorName, brandName: "", brandId: data.brandId, hexColor: result.thread.hexColor ?? "#000000" };
    }
    if (type === "BEAD") {
      const result = await createBead({ name: data.name, productCode: data.code, brandId: data.brandId });
      if (!result.success) throw new Error(result.error);
      return { id: result.bead.id, type: "BEAD", code: result.bead.productCode, name: result.bead.name, brandName: "", brandId: data.brandId, hexColor: "#000000" };
    }
    const result = await createSpecialtyItem({ name: data.name, productCode: data.code, brandId: data.brandId });
    if (!result.success) throw new Error(result.error);
    return { id: result.specialtyItem.id, type: "SPECIALTY", code: result.specialtyItem.productCode, name: result.specialtyItem.name, brandName: "", brandId: data.brandId, hexColor: "#000000" };
  };
  adapterRef.current = new CreationFlowAdapter(setSupplyRows, searchFn, createFn);
}
```
