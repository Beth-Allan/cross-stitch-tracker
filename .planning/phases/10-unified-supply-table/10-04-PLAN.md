    ```typescript
    export function useSupplyTable(
      adapter: SupplyTableAdapter,
      calcParams: CalcParams,
      existingSupplyIds: Set<string>,
    ) {
      // State
      const [supplyType, setSupplyType] = useState<SupplyType>("THREAD");
      const [searchText, setSearchText] = useState("");
      const [searchResults, setSearchResults] = useState<SupplySearchResult[]>([]);
      const [isSearching, setIsSearching] = useState(false);
      const [selectedItem, setSelectedItem] = useState<SupplySearchResult | null>(null);
      const [stitchCount, setStitchCountRaw] = useState(0);
      const [need, setNeed] = useState(1);
      const [isAutoCalc, setIsAutoCalc] = useState(true);
      const [showCreateDialog, setShowCreateDialog] = useState(false);
      const [createSearchText, setCreateSearchText] = useState("");

      // Debounced search -- 150ms delay per RESEARCH.md Pitfall 4
      useEffect for debounced search...

      // Auto-calc -- per D-07, use real calculateSkeins
      function setStitchCount(value: number) {
        setStitchCountRaw(value);
        if (supplyType === "THREAD" && isAutoCalc && value > 0) {
          const calculated = calculateSkeins({
            stitchCount: value,
            strandCount: calcParams.strandCount,
            fabricCount: calcParams.fabricCount,
            overCount: calcParams.overCount,
            wastePercent: calcParams.wastePercent,
          });
          setNeed(calculated);
        }
      }

      // Manual need override
      function setNeedManual(value: number) {
        setNeed(value);
        setIsAutoCalc(false);
      }

      // Select item from autocomplete
      function selectItem(item: SupplySearchResult) {
        setSelectedItem(item);
        setSearchText("");
        setSearchResults([]);
        // Reset qty fields to defaults
        setStitchCountRaw(0);
        setNeed(1);
        setIsAutoCalc(true);
      }

      // Commit add row
      async function commitRow(): Promise<{ success: boolean; focusTarget: "search" }> {
        if (!selectedItem) return { success: false, focusTarget: "search" };

        const effectiveNeed = need > 0 ? need : 1;
        let result: Result;

        switch (supplyType) {
          case "THREAD":
            result = await adapter.addThread(selectedItem.id, stitchCount, effectiveNeed);
            break;
          case "BEAD":
            result = await adapter.addBead(selectedItem.id, stitchCount, effectiveNeed);
            break;
          case "SPECIALTY":
            result = await adapter.addSpecialty(selectedItem.id, effectiveNeed);
            break;
        }

        if (result.success) {
          // Reset for next add -- type stays sticky
          setSelectedItem(null);
          setSearchText("");
          setStitchCountRaw(0);
          setNeed(1);
          setIsAutoCalc(true);
          return { success: true, focusTarget: "search" };
        }
        return { success: false, focusTarget: "search" };
      }

      // Reset all add-row state
      function resetAddRow() {
        setSelectedItem(null);
        setSearchText("");
        setSearchResults([]);
        setStitchCountRaw(0);
        setNeed(1);
        setIsAutoCalc(true);
      }

      // Handle inline create (per D-03)
      async function handleCreateSupply(data: CreateSupplyData): Promise<void> {
        const created = await adapter.createSupply(supplyType, data);
        selectItem(created);
        setShowCreateDialog(false);
      }

      // Return value with focusTarget hint for the add-row component
      return {
        supplyType, setSupplyType,
        searchText, setSearchText,
        searchResults, isSearching,
        selectedItem, selectItem,
        stitchCount, setStitchCount,
        need, setNeedManual, isAutoCalc,
        commitRow, resetAddRow,
        showCreateDialog, setShowCreateDialog, createSearchText, setCreateSearchText,
        handleCreateSupply,
        getFocusTarget: () => {
          if (!selectedItem) return "search";
          if (supplyType === "SPECIALTY") return "need";
          return "stitches";
        },
      };
    }
    ```
