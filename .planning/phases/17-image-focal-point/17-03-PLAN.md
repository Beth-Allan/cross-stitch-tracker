2. Create `src/components/features/charts/project-detail/crop-guide-overlay.tsx`:
   ```typescript
   "use client";

   interface CropGuideOverlayProps {
     focalX: number; // 0-1 normalized
     focalY: number; // 0-1 normalized
     containerWidth: number; // px
     containerHeight: number; // px
   }

   export function CropGuideOverlay({
     focalX,
     focalY,
     containerWidth,
     containerHeight,
   }: CropGuideOverlayProps) {
     // 60% of container width, max 360px
     const guideWidth = Math.min(containerWidth * 0.6, 360);
     const guideHeight = guideWidth * (3 / 4); // 4:3 aspect ratio

     // Center on focal point, clamp to bounds
     let left = focalX * containerWidth - guideWidth / 2;
     let top = focalY * containerHeight - guideHeight / 2;
     left = Math.max(0, Math.min(containerWidth - guideWidth, left));
     top = Math.max(0, Math.min(containerHeight - guideHeight, top));

     return (
       <>
         {/* Dimming overlay — covers entire container */}
         <div className="pointer-events-none absolute inset-0 z-10 bg-black/50" />
         {/* Clear window (crop guide area) — cuts through the dimming */}
         <div
           className="pointer-events-none absolute z-10 border-2 border-dashed border-white/80 transition-all duration-100 ease-out"
           style={{
             left: `${left}px`,
             top: `${top}px`,
             width: `${guideWidth}px`,
             height: `${guideHeight}px`,
             boxShadow: `0 0 0 9999px rgba(0, 0, 0, 0.5)`,
             background: "transparent",
           }}
         >
           {/* Label */}
           <span className="absolute top-1 left-1 text-[11px] font-semibold uppercase tracking-wider text-white">
             Gallery preview
           </span>
         </div>
       </>
     );
   }
   ```
   Per UI-SPEC: 4:3 aspect, 60% width (max 360px), centered on focal point, clamped to bounds, dashed white/80 border, black/50 dimming outside, "Gallery preview" 11px semibold uppercase label inside top-left, 100ms ease-out transition.

   Implementation note: Use the `boxShadow: 0 0 0 9999px rgba(0,0,0,0.5)` trick on the guide rectangle instead of a separate full overlay + clip-path. This creates the "window in the dimming" effect with a single element. Remove the separate dimming div if using this approach (the shadow IS the dimming). Final choice: use the boxShadow approach for simplicity — one div does both the border and the dimming.

   ```typescript
   function handleImageClick(event: React.MouseEvent<HTMLDivElement>) {
     if (!isEditMode || !containerRef.current) return;
     const rect = containerRef.current.getBoundingClientRect();
     const x = Math.max(0, Math.min(1, (event.clientX - rect.left) / rect.width));
     const y = Math.max(0, Math.min(1, (event.clientY - rect.top) / rect.height));
     setPendingPoint({ x, y });
   }
   ```

   Save handler (per RESEARCH.md Pattern 4):
   ```typescript
   function handleSave() {
     if (!pendingPoint) return;
     startTransition(async () => {
       try {
         const result = await updateFocalPoint(chartId, pendingPoint.x, pendingPoint.y);
         if (result.success) {
           setIsEditMode(false);
           toast.success("Focal point saved");
         } else {
           toast.error("Couldn't save focal point. Try again.");
         }
       } catch {
         toast.error("Couldn't save focal point. Try again.");
       }
     });
   }
   ```

   Cancel handler:
   ```typescript
   function handleCancel() {
     setPendingPoint(
       initialFocalPointX != null && initialFocalPointY != null
         ? { x: initialFocalPointX, y: initialFocalPointY }
         : null
     );
     setIsEditMode(false);
   }
   ```

   Reset handler:
   ```typescript
   function handleReset() {
     startTransition(async () => {
       try {
         const result = await updateFocalPoint(chartId, null, null);
         if (result.success) {
   ```
