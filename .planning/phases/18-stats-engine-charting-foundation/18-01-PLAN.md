    **Step 2: Implement src/lib/queries/stats/timezone.ts (GREEN)**

    Per D-04 and D-05, create the timezone abstraction:

    ```typescript
    import { TZDate } from "@date-fns/tz";
    import { startOfDay, endOfDay, startOfWeek, startOfMonth, startOfYear } from "date-fns";
    import type { LocalDateBoundaries } from "@/types/stats";

    export function getUserTimezone(_userId: string): string {
      return process.env.STATS_TIMEZONE ?? "America/Denver";
    }

    export function getLocalDayBoundaries(timezone: string): LocalDateBoundaries {
      const now = TZDate.tz(timezone);
      return {
        todayStart: startOfDay(now),
        todayEnd: endOfDay(now),
        weekStart: startOfWeek(now),
        monthStart: startOfMonth(now),
        yearStart: startOfYear(now),
      };
    }
    ```

    The function accepts a timezone string and returns Date objects representing boundaries in that timezone. The TZDate constructor creates a "now" in the specified timezone, and date-fns functions compute boundaries respecting that timezone context.
