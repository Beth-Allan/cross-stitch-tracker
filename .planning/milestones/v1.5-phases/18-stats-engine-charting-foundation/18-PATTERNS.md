### `src/lib/queries/stats/timezone.ts` (utility, transform)

**Analog:** `src/lib/auth-guard.ts`

**Pattern: thin abstraction wrapping env/config** (lines 1-22):
```typescript
import { auth } from "@/lib/auth";

export type AuthUser = {
  id: string;
  name?: string | null;
  email?: string | null;
  image?: string | null;
};

export async function requireAuth(): Promise<AuthUser> {
  const session = await auth();
  if (!session?.user?.id) throw new Error("Unauthorized");
  return session.user as AuthUser;
}
```

**Apply this pattern:** `getUserTimezone(userId)` follows the same shape -- a thin function that reads config (env var now, DB later) and returns a typed value. Export the return type. Keep it synchronous for now since it just reads `process.env`.

**No "use server" needed:** This is a pure utility, not a server action. No auth guard needed -- it receives `userId` as a parameter from the caller (which already authenticated).

**Stats page will call `requireAuth()` directly** (unlike dashboard which delegates to action functions), then pass `userId` to query functions.

### `src/components/features/stats/stats-page-shell.tsx` (component, client)

**Analog:** `src/components/features/dashboard/dashboard-tabs.tsx`

**Full pattern** (lines 1-51):
```typescript
"use client";

import type { ReactNode } from "react";
import { useQueryState, parseAsStringLiteral } from "nuqs";
import { Tabs, TabsList, TabsTrigger, TabsContent } from "@/components/ui/tabs";

export const DASHBOARD_TABS = ["library", "progress"] as const;
export type DashboardTab = (typeof DASHBOARD_TABS)[number];

const TAB_CONFIG = [
  { value: "library" as const, label: "Your Library" },
  { value: "progress" as const, label: "Progress" },
] as const;

interface DashboardTabsProps {
  libraryContent: ReactNode;
  progressContent: ReactNode;
}

export function DashboardTabs({ libraryContent, progressContent }: DashboardTabsProps) {
  const [tab, setTab] = useQueryState(
    "tab",
    parseAsStringLiteral([...DASHBOARD_TABS]).withDefault("library"),
  );

  const contentMap: Record<DashboardTab, ReactNode> = {
    library: libraryContent,
    progress: progressContent,
  };

  return (
    <Tabs value={tab} onValueChange={(val) => setTab(val as DashboardTab)}>
      <TabsList variant="line">
        {TAB_CONFIG.map(({ value, label }) => (
          <TabsTrigger key={value} value={value} className="min-h-11">
            {label}
          </TabsTrigger>
        ))}
      </TabsList>
      {DASHBOARD_TABS.map((tabValue) => (
        <TabsContent key={tabValue} value={tabValue} className="pt-6">
          {contentMap[tabValue]}
        </TabsContent>
      ))}
    </Tabs>
  );
}
```

**Key patterns:**
1. `"use client"` at top
2. `nuqs` for URL-synced tab state
3. `ReactNode` children passed from Server Component parent
4. Exported tab constants for testing
5. `Tabs`/`TabsList`/`TabsTrigger`/`TabsContent` from `@/components/ui/tabs`
